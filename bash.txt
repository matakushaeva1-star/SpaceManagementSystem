.venv\Scripts\activate --> активация venv
pip install -r requirements.txt --> Проверка и установка зависимостей
python manage.py makemigrations --> Синхронизация структуры базы данных (Миграции)
python manage.py migrate --> Применение миграций (создание таблиц пользователей, сессий, админки)
python manage.py createsuperuser --> Создание учетной записи администратора (Суперпользователя)
python manage.py runserver --> Запуск локального веб-сервера разработки
python manage.py test --> Фреймворк автоматически ищет файлы, начинающиеся с test_*.py во всех приложениях проекта
